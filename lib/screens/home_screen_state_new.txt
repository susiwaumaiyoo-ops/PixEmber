  /// PKCE認証フローでPixivログインダイアログを表示�E�アプリ冁E��WebViewによる全自動ログイン�E�E
  ///
  /// 外部ブラウザへの遷移・URLの手動コピ�Eを廁E��し、アプリ冁E�E [WebViewWidget] に
  /// Pixivログイン画面を直接描画する。ログイン完亁E��のリダイレクト�EURLめE
  /// [NavigationDelegate] で自動横取りし、認可コーチEcode)を�E自動抽出して
  /// ト�Eクン交換API�E�E_exchangeCodeForToken]�E�を呼び出す、E
  Future<void> _showPKCELoginDialog() async {
    // PKCEコード�Eリファイアとチャレンジを生戁E
    final codeVerifier = _generateCodeVerifier();
    final codeChallenge = _generateCodeChallenge(codeVerifier);

    // 認証開始URL�E�正しいエントリURL�E�を構篁E
    // 中継用の裏URL�E�E../auth/pixiv/start�E�ではなく、ログイン画面へ直接遷移する
    // エントリポイント！Eweb/v1/login�E�を持E��する、E
    // Uri が�E動でパラメータめEURL エンコードするため、code_challenge はそ�Eまま埋め込む、E
    final authUrl = Uri.https('app-api.pixiv.net', '/web/v1/login', {
      'code_challenge': codeChallenge,
      'code_challenge_method': 'S256',
      'client': 'pixiv-android',
    });

    // バックグラウンドでアプリがリセチE��されてめEcode_verifier ぁE
    // 消失しなぁE��ぁE��ダイアログ表示「直前」に SharedPreferences へ退避
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('temp_login_code_verifier', codeVerifier);

    if (!mounted) return;

    // WebViewコントローラーを生成し、Pixivログイン画面を読み込む
    final controller = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setNavigationDelegate(
        NavigationDelegate(
          // 全ての遷移を監視し、コールバックURLを�E動横取りする
          onNavigationRequest: (NavigationRequest request) async {
            final url = request.url;

            // リダイレクト�EがコールバックURLで始まる場合、code を�E動抽出
            if (url.startsWith(
              'https://app-api.pixiv.net/web/v1/users/auth/pixiv/callback',
            )) {
              String? code;

              // 1. クエリパラメータから code を抽出
              try {
                final uri = Uri.parse(url);
                code = uri.queryParameters['code'];
                // フラグメント�Eに code が含まれる場合も老E�E
                if (code == null && uri.fragment.contains('code=')) {
                  code = Uri.parse(
                    'pixiv://x?${uri.fragment}',
                  ).queryParameters['code'];
                }
              } catch (e) {
                // 2. パ�Eスエラー時�Eフォールバック�E�正規表現�E�E
                final regExp = RegExp(r'code=([^&]+)');
                final match = regExp.firstMatch(url);
                if (match != null) {
                  code = match.group(1);
                }
              }

              print('[WebView Hook] コールバックURLを検知: $url');
              print('[WebView Hook] 抽出した code: $code');

              if (code != null && code.isNotEmpty) {
                // SharedPreferences から code_verifier を復允E��て利用
                final restoredVerifier = prefs.getString(
                  'temp_login_code_verifier',
                );
                if (restoredVerifier != null && restoredVerifier.isNotEmpty) {
                  // ダイアログを�Eログラムで自動的に閉じめE
                  if (mounted) {
                    Navigator.of(context).pop();
                  }
                  // 抽出した code と code_verifier でト�Eクン交換APIを�E動実衁E
                  _exchangeCodeForToken(code, restoredVerifier);
                } else {
                  // code_verifier が消失してぁE��場合�Eユーザーへ通知
                  if (mounted) {
                    ScaffoldMessenger.of(context).showSnackBar(
                      const SnackBar(
                        content: Text('認証セチE��ョンが�EれてぁE��す。�E度ログインし直してください、E),
                      ),
                    );
                  }
                }
              }
              // コールバックURL自体�E読み込みは防止�E�エラー画面を�EさなぁE��E
              return NavigationDecision.prevent;
            }

            // それ以外�E遷移は許可�E�ログイン画面冁E�E通常遷移�E�E
            return NavigationDecision.navigate;
          },
        ),
      )
      ..loadRequest(authUrl);

    await showDialog<void>(
      context: context,
      barrierDismissible: false,
      builder: (dialogContext) {
        return AlertDialog(
          title: const Text('Pixiv にログイン'),
          contentPadding: EdgeInsets.zero,
          content: SizedBox(
            width: double.maxFinite,
            height: 500,
            child: WebViewWidget(controller: controller),
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.of(dialogContext).pop();
              },
              child: const Text('キャンセル'),
            ),
          ],
        );
      },
    );
  }

  /// PKCEコード�Eリファイアを生成！EFC7636 準拠: 32バイト乱数 -> Base64URL・パディング無�E�E
  String _generateCodeVerifier() {
    final random = Random.secure();
    final values = List<int>.generate(32, (i) => random.nextInt(256));
    return base64Url.encode(values).replaceAll('=', '');
  }

  /// PKCEコードチャレンジを生成！EHA-256 -> Base64URL・パディング無�E�E
  String _generateCodeChallenge(String verifier) {
    final bytes = utf8.encode(verifier);
    final digest = sha256.convert(bytes);
    return base64Url.encode(digest.bytes).replaceAll('=', '');
  }

  Future<void> _loadSearchHistory() async {

      final prefs = await SharedPreferences.getInstance();
      if (!mounted) return;

    } catch (e) {

  Future<void> _saveSearchHistory(String word) async {
    if (word.trim().isEmpty) return;
    try {
      final prefs = await SharedPreferences.getInstance();
      if (!mounted) return;
      final list = List<String>.from(_searchHistory);
      list.remove(word); // 重褁E��削除
      list.insert(0, word); // 先頭に挿入
      if (list.length > 15) {
        list.removeLast(); // 最大15件
      }
      setState(() {
        _searchHistory = list;
        _showHistoryList =
            _searchFocusNode.hasFocus && _searchHistory.isNotEmpty;
      });
      await prefs.setStringList('search_history', list);
    } catch (e) {
      debugPrint('Error saving search history: $e');
    }
  }

  Future<void> _deleteSearchHistoryItem(String word) async {
    try {
      final prefs = await SharedPreferences.getInstance();
      if (!mounted) return;
      final list = List<String>.from(_searchHistory);
      list.remove(word);
      setState(() {
        _searchHistory = list;
        _showHistoryList =
            _searchFocusNode.hasFocus && _searchHistory.isNotEmpty;

// --- 無限スクロール検知 ---
  void _onScroll() {
    if (!_scrollController.hasClients) return;
    final maxScroll = _scrollController.position.maxScrollExtent;
    final currentScroll = _scrollController.position.pixels;

    // 最下部手前200pxに達したら次ペ�Eジ読み込み
    if (maxScroll - currentScroll <= 200) {
      if (!_isLoading && !_isLoadingMore && !_isFetchingNextPage && _nextOffset != null && !_rateLimited) {
        _isFetchingNextPage = true;
        _fetchNextPage().then((_) {
          if (mounted) {
            setState(() {
              _isFetchingNextPage = false;
            });
          }
        });
      }
    }
  }

// チE�Eタ新規取征E
  Future<void> _fetchData() async {
    setState(() {
      _isLoading = true;
      _errorMessage = null;
      _nextOffset = null;
      _searchItem = null;
      _allTextSearchState = null; // 全斁E��索用の状態をリセチE��
      _illusts = [];
      _novels = [];
    });

    final activeSubMode = _currentIndex == _illustIndex ? _illustSubMode : _novelSubMode;

    try {
      if (_currentIndex == _illustIndex) {
        // --- イラストデータ取征E(PixivApiService 直接呼び出ぁE ---
        final result = await _fetchIllusts(activeSubMode, 0);
        if (!mounted) return;
        setState(() {
          _illusts = result.items;
          _nextOffset = result.nextOffset; // API の next_url から安�Eに抽出
          _searchItem = result.searchItem; // 検索時�EみセチE��
          _isLoading = false;
        });
      } else if (_currentIndex == _novelIndex) {
        // --- 小説チE�Eタ取征E(PixivApiService 直接呼び出ぁE ---
        final result = await _fetchNovels(activeSubMode, 0);
        if (!mounted) return;
        setState(() {
          _novels = result.items;
          _nextOffset = result.nextOffset;
          _searchItem = result.searchItem;
          _isLoading = false;
        });
        // シリーズ統計文字数フィルターが有効なら非同期で合計文字数を集訁E
        await _computeSeriesTextLengths(result.items);
      } else {
        // フィーリング発掘タチE ローカルDBからの読み込みのため特別な処琁E�E不要E
        // FeelingDiscoveryScreen が�E前でチE�Eタを取征E
        if (!mounted) return;
        setState(() {
          _isLoading = false;
        });
      }
    } catch (e) {
      if (!mounted) return;
      final errorStr = e.toString();
      if (errorStr.contains('429') || errorStr.contains('rate limit') || errorStr.contains('レート制陁E)) {
        setState(() {
          _errorMessage = 'Pixivのアクセス制限（レート制限）が発生しました。しばらく時間を置ぁE��から再試行してください、E;
          _isLoading = false;
        });
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('Pixivのアクセス制限（レート制限）が発生しました。しばらく時間を置ぁE��から再試行してください、E),
              duration: Duration(seconds: 5),
            ),
          );
        }
      } else {
        setState(() {
          _errorMessage = 'チE�Eタの取得に失敗しました、En詳細: $e';
          _isLoading = false;
        });
      }
    }
  }

// 次ペ�Eジ追加読み込み (無限スクロール)
  Future<void> _fetchNextPage() async {
    if (_nextOffset == null) return;
    // フィーリング発掘タブでは無限スクロールなぁE
    if (_currentIndex == _feelingDiscoveryIndex) return;

    setState(() {
      _isLoadingMore = true;
      _rateLimited = false;
    });

    final activeSubMode = _currentIndex == _illustIndex ? _illustSubMode : _novelSubMode;

    try {
      if (_currentIndex == _illustIndex) {
        // --- イラスト次ペ�Eジ ---
        final result = await _fetchIllusts(activeSubMode, _nextOffset!);
        if (!mounted) return;
        setState(() {
          _illusts.addAll(result.items);
          _nextOffset = result.nextOffset; // さらに次があれ�E更新、なければ null
          _isLoadingMore = false;
        });
      } else if (_currentIndex == _novelIndex) {
        // --- 小説次ペ�Eジ ---
        final result = await _fetchNovels(activeSubMode, _nextOffset!);
        if (!mounted) return;
        // 全斁E��索は褁E��ターゲチE��を統合するため、既存IDと重褁E��る作品を除外して全件を確実に取征E
        final existingIds = _novels.map((n) => n.id).toSet();
        final newItems = result.items
            .where((n) => !existingIds.contains(n.id))
            .toList();
        setState(() {
          _novels.addAll(newItems);
          _nextOffset = result.nextOffset;
          _isLoadingMore = false;
        });
        // 新たに追加された�Eのシリーズ合計文字数を集訁E
        await _computeSeriesTextLengths(result.items);
      }
    } catch (e) {
      debugPrint('Error fetching next page: $e');
      final errorStr = e.toString();
      if (errorStr.contains('429') || errorStr.contains('rate limit') || errorStr.contains('レート制陁E)) {
        if (!mounted) return;
        setState(() {
          _rateLimited = true;
          _isLoadingMore = false;
        });
        // SnackBar通知
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('Pixivのアクセス制限（レート制限）が発生しました。しばらく時間を置ぁE��から再試行してください、E),
              duration: Duration(seconds: 5),
            ),
          );
        }
      } else {
        if (!mounted) return;
        setState(() {
          _isLoadingMore = false;
        });
      }
    }
  }

  // PixivApiService を呼び出してイラスト一覧を取征E(サーバ�Eレス)
  Future<FetchResult<Illust>> _fetchIllusts(int subMode, int offset) async {
    if (subMode == 0) {
      return _pixivApiService.getRecommend(offset: offset);
    } else if (subMode == 1) {

    }
  }

  // PixivApiService を呼び出して小説一覧を取征E(サーバ�Eレス)
  Future<FetchResult<Novel>> _fetchNovels(int subMode, int offset) async {
    // 斁E��数フィルタ�E�Ehort/medium/long/custom�E�を minLen/maxLen に正規化、E
    // 検索モードだけでなくおすすめ�Eランキングでもサーバ�Eサイドに渡す、E
    int? minLen;
    int? maxLen;
    if (_selectedNovelTextLengthLimit == 'short') {
      maxLen = 5000;
    } else if (_selectedNovelTextLengthLimit == 'medium') {
      minLen = 5000;
      maxLen = 20000;
    } else if (_selectedNovelTextLengthLimit == 'long') {
      minLen = 20000;
    } else if (_selectedNovelTextLengthLimit == 'custom') {
      minLen = int.tryParse(_minTextLengthController?.text ?? '');
      maxLen = int.tryParse(_maxTextLengthController?.text ?? '');
    }

    if (subMode == 0) {
      return _pixivApiService.getNovelRecommend(
        offset: offset,
        startTextLength: minLen,
        endTextLength: maxLen,
      );
    } else if (subMode == 1) {
      if (_selectedNovelSearchTarget == 'all_text') {
        // 全斁E��索モーチE AllTextSearchState を使って個別のペ�Eジングを管琁E
        final state = _allTextSearchState;
        final result = await _pixivApiService.searchNovelAllText(
          _currentSearchWord,
          _selectedSort,
          _selectedNovelAgeLimit,
          minLen,
          maxLen,
          bookmarkFilter: _selectedNovelBookmarkFilter,
          state: state,
        );
        // 更新された状態を保孁E
        _allTextSearchState = result.state;
        return result.result;
      }
      return _pixivApiService.searchNovel(
        _currentSearchWord,
        _selectedNovelSearchTarget,
        _selectedSort,
        offset,
        _selectedNovelAgeLimit,
        minLen,
        maxLen,
        bookmarkFilter: _selectedNovelBookmarkFilter,
      );
    } else {
      return _pixivApiService.getNovelRanking(
        _selectedNovelRankMode,
        offset: offset,
        startTextLength: minLen,
        endTextLength: maxLen,
      );
    }
  }

  // シリーズ統計文字数フィルター用: 吁E��説の seriesId からシリーズ全話の
  // textLength を合算し _seriesTotalTextLengthCache にキャチE��ュする、E
  // すでにキャチE��ュ済みの seriesId はスキチE�Eする (N+1 呼び出し抑制)、E
  Future<void> _computeSeriesTextLengths(List<Novel> novels) async {
    if (_selectedNovelSeriesTextLengthLimit == 'all') return;
    final targets = novels
        .where(
          (n) =>
              n.series != null &&
              n.series!.id != 0 &&
              !_seriesTotalTextLengthCache.containsKey(n.series!.id),
        )
        .toList();
    if (targets.isEmpty) return;

    final newValues = <int, int>{};
    for (final novel in targets) {
      final seriesId = novel.series!.id;
      try {
        final episodes = await _pixivApiService.getNovelSeriesAll(seriesId);
        final total = episodes.fold<int>(0, (sum, ep) => sum + ep.textLength);
        newValues[seriesId] = total;
      } catch (e) {
        debugPrint('シリーズ統計文字数の雁E��に失敁E(seriesId=$seriesId): $e');
        // 失敗時は 0 として扱ぁE���E試行しなぁE��ぁE��ャチE��ュしておく
        newValues[seriesId] = 0;
      }
    }

    if (!mounted) return;
    // 雁E��結果を一度に結合してから setState で再描画�E��E度再描画を避ける�E�E
    final merged = {..._seriesTotalTextLengthCache, ...newValues};
    setState(() {
      _seriesTotalTextLengthCache
        ..clear()
        ..addAll(merged);
    });
  }

  // シリーズ統計文字数フィルターの閾値を取征E(short/medium/long/custom 対忁E
  _SeriesLengthThreshold _getSeriesLengthThreshold() {
    int? minLen;

    if (_selectedNovelSeriesTextLengthLimit == 'short') {

    }
    return _SeriesLengthThreshold(minLen: minLen, maxLen: maxLen);
  }

  Widget _buildIllustGrid(int crossAxisCount) {
    final filteredIllusts = _illusts.where((illust) {
      // 1. 作品種別フィルター
      if (_selectedWorkType != 'all') {
        if (_selectedWorkType == 'illust' && illust.type != 'illust') {
          return false;
        }
        if (_selectedWorkType == 'manga' && illust.type != 'manga') {
          return false;
        }
        if (_selectedWorkType == 'ugoira' && illust.type != 'ugoira') {
          return false;
        }
        if (_selectedWorkType == 'novel') {
          return false;
        }
      }
      // 2. 年齢制限フィルター
      final hasR18Tag = illust.tags.any(
        (t) =>
            t.toLowerCase().contains('r-18') || t.toLowerCase().contains('r18'),
      );
      if (_selectedAgeLimit == 'safe' && hasR18Tag) return false;
      if (_selectedAgeLimit == 'r18' && !hasR18Tag) return false;
      return true;
    }).toList();

    if (filteredIllusts.isEmpty && _illusts.isNotEmpty) {
      return const Center(
        child: Text(
          'フィルターに一致するイラストが見つかりませんでした、E,
          style: TextStyle(color: Colors.grey),
        ),
      );
    }

    if (_illusts.isEmpty) {
      return const Center(
        child: Text('イラストが見つかりませんでした、E, style: TextStyle(color: Colors.grey)),
      );
    }

    return RefreshIndicator(
      onRefresh: _fetchData,
      child: GridView.builder(
        controller: _scrollController,
        physics: const ClampingScrollPhysics(),
        padding: const EdgeInsets.all(6.0),
        gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: crossAxisCount,
          crossAxisSpacing: 6.0,
          mainAxisSpacing: 6.0,
          childAspectRatio: 0.75,
        ),
itemCount: filteredIllusts.length + (_nextOffset != null ? 1 : 0),
        itemBuilder: (context, index) {
          // 最下部に到達した時のローチE��ング追加表示
          if (index == filteredIllusts.length) {
            if (_rateLimited) {
              return Center(
                child: Padding(
                  padding: const EdgeInsets.all(16.0),
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      const Text(
                        'アクセス制限が発生しました',
                        style: TextStyle(color: Colors.red, fontWeight: FontWeight.bold),
                      ),
                      const SizedBox(height: 8),
                      const Text(
                        'Pixivのアクセス制限（レート制限）が発生しました、Enし�Eらく時間を置ぁE��から再試行してください、E,
                        textAlign: TextAlign.center,
                        style: TextStyle(color: Colors.grey),
                      ),
                      const SizedBox(height: 16),
                      ElevatedButton.icon(
                        onPressed: _fetchNextPage,
                        icon: const Icon(Icons.refresh),
                        label: const Text('再試衁E),
                      ),
                    ],
                  ),
                ),
              );
            }
            return const Center(
              child: Padding(
                padding: EdgeInsets.all(16.0),
                child: CircularProgressIndicator(color: Colors.pinkAccent),
              ),
            );
          }

          final illust = filteredIllusts[index];
          return Card(
            clipBehavior: Clip.antiAlias,
            elevation: 3,
            child: InkWell(
              onTap: () async {
                // 詳細画面へ遷移
                await Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (context) => IllustDetailScreen(
                      illust: illust,
                      onTagTap: _onTagSelected,
                      onBookmarkChanged: (newVal) {
                        setState(() {
                          illust.isBookmarked = newVal;
                        });
                      },
                    ),
                  ),
                );
                // 戻った際、状態整合性を保つため再描画

  Widget _buildNovelList() {
    final filteredNovels = _novels.where((novel) {
      // 1. 年齢制限フィルター (小説専用)
      final hasR18Tag = novel.tags.any(
        (t) =>
            t.toLowerCase().contains('r-18') || t.toLowerCase().contains('r18'),
      );
      if (_selectedNovelAgeLimit == 'safe' && hasR18Tag) {
        return false;
      }
      if (_selectedNovelAgeLimit == 'r18' && !hasR18Tag) {
        return false;
      }

      // 2. 本斁E�E長さフィルター (小説専用)
      if (_selectedNovelTextLengthLimit == 'short' && novel.textLength > 5000) {
        return false;
      }
      if (_selectedNovelTextLengthLimit == 'medium' &&
          (novel.textLength <= 5000 || novel.textLength > 20000)) {
        return false;
      }
      if (_selectedNovelTextLengthLimit == 'long' &&
          novel.textLength <= 20000) {
        return false;
      }
      if (_selectedNovelTextLengthLimit == 'custom') {
        final minLen = int.tryParse(_minTextLengthController?.text ?? '');
        final maxLen = int.tryParse(_maxTextLengthController?.text ?? '');
        if (minLen != null && novel.textLength < minLen) {
          return false;
        }
        if (maxLen != null && novel.textLength > maxLen) {
          return false;
        }
      }

      // 2.5 シリーズ統計文字数フィルター (シリーズ全体�E合計文字数)
      if (_selectedNovelSeriesTextLengthLimit != 'all') {
        // シリーズでなぁE��品は常に除夁E
        if (novel.series == null || novel.series!.id == 0) {
          return false;
        }
        final seriesTotal = _seriesTotalTextLengthCache[novel.series!.id];
        if (seriesTotal != null) {
          final threshold = _getSeriesLengthThreshold();
          if (threshold.minLen != null && seriesTotal < threshold.minLen!) {
            return false;
          }
          if (threshold.maxLen != null && seriesTotal > threshold.maxLen!) {
            return false;
          }
        }
        // 雁E��未完亁E(キャチE��ュ未存在) の場合�E一旦表示し、E
        // _computeSeriesTextLengths の setState による再描画で正しく絞り込む、E

      }

      // 3. AI作品フィルター (アプリ冁E��ーカル)
      // novel_ai_type ぁE2 の場合に AI生�E作品と判宁E
      final bool isAi = novel.aiType == 2;
      if (_selectedNovelAiFilter == 'hide' && isAi) {
        return false;
      }
      if (_selectedNovelAiFilter == 'only' && !isAi) {
        return false;
      }

      // 4. シリーズ作品のみフィルター (アプリ冁E��ーカル)
      if (_novelSeriesOnly) {
        final hasSeries = novel.series != null && novel.series!.id != 0;
        if (!hasSeries) return false;
      }

      // 5. 除外タグフィルター (アプリ冁E��ーカル)
      if (_novelExcludeTags.isNotEmpty) {
        final lowerTags = novel.tags.map((t) => t.toLowerCase()).toSet();
        for (final ex in _novelExcludeTags) {
          if (lowerTags.contains(ex.toLowerCase())) {
            return false;
          }
        }
      }

      return true;
    }).toList();

    // 表示寁E��の高さ�E�Eompact=95px / comfortable=135px�E�E
    final int itemHeight = _novelDensityMode == 'compact' ? 95 : 135;

      return const Center(
        child: Text(
          'フィルターに一致する小説が見つかりませんでした、E,
          style: TextStyle(color: Colors.grey),
        ),
      );
    }

    if (_novels.isEmpty) {
      return const Center(
        child: Text('小説が見つかりませんでした、E, style: TextStyle(color: Colors.grey)),
      );
    }

      onRefresh: _fetchData,

  // 数値フォーマット�Eルパ�E (3桁カンマ区刁E��)
  String _formatNumber(int? number) {
    if (number == null) return '0';
    final str = number.toString();
    final reg = RegExp(r'(\d{1,3})(?=(\d{3})+(?!\d))');
    return str.replaceAllMapped(reg, (Match m) => '${m[1]},');
  }

  // 読書中�E�しおり�E��E小説の簡易情報を取得すめE
  Future<List<Map<String, dynamic>>> _getRecentBookmarks() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final List<String> bookmarkedIds =
          prefs.getStringList('novel_bookmark_ids') ?? [];
      final List<Map<String, dynamic>> results = [];

      for (final idStr in bookmarkedIds) {
        final id = int.tryParse(idStr);
        if (id == null) continue;

        final title = prefs.getString('novel_title_$id');
        final author = prefs.getString('novel_author_$id');
        final progress = prefs.getDouble('novel_progress_$id') ?? 0.0;
        final lastRead = prefs.getInt('novel_last_read_$id') ?? 0;

        if (title != null && progress > 0.0 && progress < 100.0) {
          // 読亁E100%)してぁE��ぁE��のを対象
          results.add({
            'id': id,
            'title': title,
            'author': author ?? '不�E',
            'progress': progress,
            'lastRead': lastRead,
          });
        }
      }

      // 最終読書日時！EastRead�E�が新しい頁E��ソーチE
      results.sort((a, b) => b['lastRead'].compareTo(a['lastRead']));

      return results.take(5).toList();
    } catch (e) {
      debugPrint('しおり履歴の取得失敁E $e');
      return [];
    }
  }

Future<void> _handleGoogleBackup() async {
    setState(() => _isBackingUp = true);
    _showLoadingDialog();
    try {
      final success = await _driveService.backupJSON();
      _hideLoadingDialog();
      setState(() => _isBackingUp = false);
      if (success && mounted) {
        final now = DateTime.now().toIso8601String();
        final prefs = await SharedPreferences.getInstance();
        await prefs.setString('GOOGLE_DRIVE_LAST_SYNC', now);
        setState(() => _lastSyncTimestamp = now);
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(const SnackBar(content: Text('バックアチE�E完亁E��ました�E�E)));
      }
    } catch (e) {
      _hideLoadingDialog();
      setState(() => _isBackingUp = false);
      if (mounted) {
        ScaffoldMessenger.of(
          context,
        ).showSnackBar(SnackBar(content: Text('バックアチE�E失敁E $e')));
      }
    }
  }

Future<void> _handleGoogleRestore() async {
    setState(() => _isRestoring = true);
    _showLoadingDialog();

      final summary = await _driveService.restoreJSON();
      _hideLoadingDialog();

      if (summary != null && mounted) {
        final now = DateTime.now().toIso8601String();
